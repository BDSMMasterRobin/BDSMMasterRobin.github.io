function encode(data) { // Kompaktes Datenformat: // [Name, Geburtsdatum, Geschlecht, X, Instagram, Reddit, Fet, Recon, Bluesky, Vorlieben] // // Vorlieben: // [FrageIndex, AuswahlIndex, FrageIndex, AuswahlIndex, ...] // // Dadurch müssen weder die langen Fragestellungen noch die langen // Auswahltexte in den Link geschrieben werden. const preferences = []; Object.entries(data.preferences || {}).forEach(([questionIndex, choice]) => { const choiceIndex = preferenceChoices.indexOf(choice); if ( /^\d+$/.test(questionIndex) && choiceIndex >= 0 && Number(questionIndex) < preferenceQuestions.length ) { preferences.push(Number(questionIndex), choiceIndex); } }); const compact = [ data.name || '', data.birthdate || '', data.gender || '', data.x || '', data.instagram || '', data.reddit || '', data.fet || '', data.recon || '', data.bluesky || '', preferences ]; return LZString.compressToEncodedURIComponent(JSON.stringify(compact)); } function decode(value) { const compact = JSON.parse( LZString.decompressFromEncodedURIComponent(value) ); if (!Array.isArray(compact)) { throw new Error('Ungültiges Datenformat'); } const [ name = '', birthdate = '', gender = '', x = '', instagram = '', reddit = '', fet = '', recon = '', bluesky = '', preferenceArray = [] ] = compact; const preferences = {}; for (let i = 0; i + 1 < preferenceArray.length; i += 2) { const questionIndex = Number(preferenceArray[i]); const choiceIndex = Number(preferenceArray[i + 1]); if ( Number.isInteger(questionIndex) && Number.isInteger(choiceIndex) && questionIndex >= 0 && questionIndex < preferenceQuestions.length && choiceIndex >= 0 && choiceIndex < preferenceChoices.length ) { preferences[questionIndex] = preferenceChoices[choiceIndex]; } } return { name, birthdate, gender, x, instagram, reddit, fet, recon, bluesky, preferences }; }
